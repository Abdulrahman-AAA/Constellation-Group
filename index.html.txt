const canvas=document.getElementById('stars-canvas'),ctx=canvas.getContext('2d');
let stars=[],meteors=[],W,H;
function resize(){W=canvas.width=window.innerWidth;H=canvas.height=window.innerHeight;}
function initStars(){stars=[];const n=Math.floor(W*H/5000);for(let i=0;i<n;i++)stars.push({x:Math.random()*W,y:Math.random()*H,r:Math.random()*1.2+0.2,alpha:Math.random()*0.6+0.2,speed:Math.random()*0.003+0.001,phase:Math.random()*Math.PI*2});}
function spawnMeteor(){meteors.push({x:Math.random()*W*0.8,y:Math.random()*H*0.3,vx:4+Math.random()*3,vy:2+Math.random()*2,alpha:1,life:0});}
function drawFrame(ts){
  ctx.clearRect(0,0,W,H);
  stars.forEach(s=>{const a=s.alpha*(0.7+0.3*Math.sin(ts*s.speed+s.phase));ctx.beginPath();ctx.arc(s.x,s.y,s.r,0,Math.PI*2);ctx.fillStyle=`rgba(232,207,138,${a})`;ctx.fill();});
  meteors=meteors.filter(m=>m.alpha>0.01);
  meteors.forEach(m=>{m.x+=m.vx;m.y+=m.vy;m.life++;m.alpha=Math.max(0,1-m.life/60);const g=ctx.createLinearGradient(m.x,m.y,m.x-m.vx*12,m.y-m.vy*12);g.addColorStop(0,`rgba(232,207,138,${m.alpha})`);g.addColorStop(1,'rgba(232,207,138,0)');ctx.beginPath();ctx.moveTo(m.x,m.y);ctx.lineTo(m.x-m.vx*12,m.y-m.vy*12);ctx.strokeStyle=g;ctx.lineWidth=1.5;ctx.stroke();});
  requestAnimationFrame(drawFrame);
}
resize();initStars();window.addEventListener('resize',()=>{resize();initStars();});
setInterval(spawnMeteor,4000);requestAnimationFrame(drawFrame);
