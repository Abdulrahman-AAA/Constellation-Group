<script>
const STORAGE_KEY='cag_site_data';
const CRED_KEY='cag_admin_creds';
function defaultCreds(){return{user:'admin',pass:'constellation2025'};}
function getCreds(){try{return JSON.parse(localStorage.getItem(CRED_KEY))||defaultCreds();}catch(e){return defaultCreds();}}
function getSiteData(){try{return JSON.parse(localStorage.getItem(STORAGE_KEY))||{};}catch(e){return {};}}
function saveSiteData(d){localStorage.setItem(STORAGE_KEY,JSON.stringify(d));}

function doLogin(){
  const u=document.getElementById('lu').value.trim();
  const p=document.getElementById('lp').value;
  const c=getCreds();
  if(u===c.user&&p===c.pass){
    document.getElementById('login-screen').style.display='none';
    const as=document.getElementById('admin-screen');
    as.style.display='flex';as.style.flexDirection='column';
    loadAllData();
  } else {
    document.getElementById('lerr').style.display='block';
    document.getElementById('lp').value='';
  }
}
document.getElementById('lu').addEventListener('keydown',e=>{if(e.key==='Enter')doLogin();});
function doLogout(){
  document.getElementById('admin-screen').style.display='none';
  document.getElementById('login-screen').style.display='flex';
  document.getElementById('lu').value='';document.getElementById('lp').value='';
}

  document.querySelectorAll('.nav-item').forEach(n=>{
    if(n.getAttribute('onclick')&&n.getAttribute('onclick').includes("'"+id+"'"))n.classList.add('active');
  });
}

function updateDashStats(d){
  const photos=d.photos||[];
  document.getElementById('stat-photos').textContent=photos.length;
  const b=document.getElementById('photo-count-badge');
  if(b)b.textContent=photos.length?photos.length+' photo'+(photos.length===1?'':'s'):'';

function handleFiles(files){
  const d=getSiteData();d.photos=d.photos||[];
  let pending=files.length;
  Array.from(files).forEach(file=>{
    const r=new FileReader();
    r.onload=e=>{
      d.photos.push({src:e.target.result,title:'',vehicle:'',id:Date.now()+Math.random()});
      pending--;
      if(pending===0){saveSiteData(d);renderPhotos(d.photos);updateDashStats(d);showToast(files.length+' photo'+(files.length===1?'':'s')+' uploaded','ok');}
    };
    r.readAsDataURL(file);
  });
}

function renderPhotos(photos){
  const grid=document.getElementById('photos-grid');
  const empty=document.getElementById('empty-msg');
  if(!photos||photos.length===0){grid.innerHTML='';if(empty){empty.style.display='block';grid.appendChild(empty);}return;}
  if(empty)empty.style.display='none';
  grid.innerHTML='';
  photos.forEach((p,i)=>{
    const card=document.createElement('div');card.className='photo-card';card.dataset.idx=i;card.draggable=true;
    card.innerHTML=`<div class="photo-badge">#${i+1}</div><img class="photo-thumb" src="${p.src}" alt="Photo ${i+1}"><div class="photo-body"><input type="text" placeholder="Title (e.g. BMW 7 Series)" value="${p.title||''}" oninput="updatePhotoField(${i},'title',this.value)"><input type="text" placeholder="Vehicle (e.g. 2023 Rolls-Royce)" value="${p.vehicle||''}" oninput="updatePhotoField(${i},'vehicle',this.value)"></div><div class="photo-actions"><div class="drag-handle" title="Drag to reorder">⠿</div><button class="btn btn-xs btn-outline" onclick="movePhoto(${i},-1)">↑</button><button class="btn btn-xs btn-outline" onclick="movePhoto(${i},1)">↓</button><button class="btn btn-xs btn-danger photo-del" onclick="deletePhoto(${i})">✕ Remove</button></div>`;
    grid.appendChild(card);
  });
  initDrag();
}

function updatePhotoField(idx,field,val){const d=getSiteData();if(d.photos&&d.photos[idx])d.photos[idx][field]=val;saveSiteData(d);}
function deletePhoto(idx){const d=getSiteData();d.photos.splice(idx,1);saveSiteData(d);renderPhotos(d.photos);updateDashStats(d);showToast('Photo removed','ok');}
function movePhoto(idx,dir){const d=getSiteData();const arr=d.photos;const ni=idx+dir;if(ni<0||ni>=arr.length)return;[arr[idx],arr[ni]]=[arr[ni],arr[idx]];saveSiteData(d);renderPhotos(arr);}

let dragSrc=null;
function initDrag(){
  document.querySelectorAll('.photo-card').forEach(card=>{
    card.addEventListener('dragstart',e=>{dragSrc=card;card.style.opacity='0.4';e.dataTransfer.effectAllowed='move';});
    card.addEventListener('dragend',()=>{card.style.opacity='1';dragSrc=null;});
    card.addEventListener('dragover',e=>{e.preventDefault();e.dataTransfer.dropEffect='move';card.style.borderColor='rgba(201,168,76,0.6)';});
    card.addEventListener('dragleave',()=>{card.style.borderColor='';});
    card.addEventListener('drop',e=>{
      e.preventDefault();card.style.borderColor='';
      if(dragSrc&&dragSrc!==card){
        const fi=parseInt(dragSrc.dataset.idx),ti=parseInt(card.dataset.idx);
        const d=getSiteData();const arr=d.photos;const moved=arr.splice(fi,1)[0];arr.splice(ti,0,moved);saveSiteData(d);renderPhotos(arr);
      }
    });
  });
}

const dz=document.getElementById('drop-zone');
dz.addEventListener('dragover',e=>{e.preventDefault();dz.classList.add('drag');});
dz.addEventListener('dragleave',()=>{dz.classList.remove('drag');});
dz.addEventListener('drop',e=>{e.preventDefault();dz.classList.remove('drag');if(e.dataTransfer.files.length)handleFiles(e.dataTransfer.files);});

function changePassword(){
  const cur=document.getElementById('s-cur').value;
  const nw=document.getElementById('s-new').value;
  const conf=document.getElementById('s-conf').value;
  const c=getCreds();
  if(cur!==c.pass){showToast('Current password incorrect','err');return;}
  if(!nw){showToast('New password cannot be empty','err');return;}
  if(nw!==conf){showToast('Passwords do not match','err');return;}
  c.pass=nw;localStorage.setItem(CRED_KEY,JSON.stringify(c));
  document.getElementById('s-cur').value='';document.getElementById('s-new').value='';document.getElementById('s-conf').value='';
  showToast('Password updated','ok');
}

  const a=document.createElement('a');a.href=URL.createObjectURL(blob);
  a.download='constellation-backup-'+new Date().toISOString().slice(0,10)+'.json';a.click();
  showToast('Backup exported','ok');
}
